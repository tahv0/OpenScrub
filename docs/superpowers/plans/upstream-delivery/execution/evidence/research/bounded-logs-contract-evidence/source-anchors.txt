  638 def worker():
  639     """Run queued jobs forever. NOTHING may end this loop: it is the
  640     server's only worker, and when it died (a SystemExit from argparse —
  641     which a thread swallows silently) the job froze at "scanning" with an
  642     empty log, Cancel was ignored and every later job queued forever until
  643     the container was restarted. Every per-job failure, BaseException
  644     included, now fails that job loudly and the loop carries on."""
  645     while True:
  646         try:
  647             item = WORK_Q.get()
  648         except BaseException:          # pragma: no cover — queue can't fail
  649             time.sleep(1)
  650             continue
  651         if item is None:               # explicit stop (tests); the watchdog
  652             WORK_Q.task_done()         # starts a fresh worker in production
  653             return
  654         try:
  655             jid, action = item[0], item[1]
  656             gen = item[2] if len(item) > 2 else None
  657             with JOBS_LOCK:
  658                 job = JOBS.get(jid)
  659                 stale = (job is None
  660                          or (gen is not None and gen != job.get("qgen", 0))
  661                          or job.get("phase") == "cancelled")
  662                 if not stale:
  663                     WORKER["current"] = jid
  664             if not stale:
  665                 _run_job(job, action)
  666         except BaseException as e:     # belt and braces: never die
  667             print(f"[web] worker: unexpected {type(e).__name__}: {e}",
  668                   flush=True)
  669         finally:
  670             WORKER["current"] = None
  671             WORK_Q.task_done()
