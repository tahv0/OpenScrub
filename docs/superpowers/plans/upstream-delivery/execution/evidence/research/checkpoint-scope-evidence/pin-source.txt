7: def atomic_report(path, report) -> None:
8:     """Stage private UTF-8 JSON beside *path*, sync, close, then replace it.
9: 
10:     A failure before replacement leaves the previous report intact and raises.
11:     Readers opening the path see a complete old or new report on filesystems
12:     supporting atomic replacement. This does not serialize read/apply/write
13:     transactions or guarantee power-loss durability: the directory is not
14:     fsynced, and filesystem/network storage guarantees vary. On Windows an open
15:     destination may prevent replacement; that failure is propagated. Temporary
16:     files use the vault's .tmp crash-cleanup convention. POSIX staging permissions
17:     are 0600; Windows access restrictions also depend on the directory ACL.
18:     """
19:     path = os.path.abspath(path)
20:     staged = None
21:     try:
22:         # NamedTemporaryFile creates exclusively with restrictive permissions.
23:         with tempfile.NamedTemporaryFile(mode="w", encoding="utf-8",
24:                                          dir=os.path.dirname(path),
25:                                          prefix=".report-", suffix=".tmp",
26:                                          delete=False) as stream:
27:             staged = stream.name
28:             json.dump(report, stream, indent=1)
29:             stream.flush()
30:             os.fsync(stream.fileno())
