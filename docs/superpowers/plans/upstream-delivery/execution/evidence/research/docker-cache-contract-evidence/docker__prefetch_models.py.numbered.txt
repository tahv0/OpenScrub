     1	#!/usr/bin/env python3
     2	"""Docker build helper: pre-fetch the auto-downloaded detection models
     3	into the image so first run works offline.
     4	
     5	Standalone ON PURPOSE — stdlib only, no openscrub import. The whole point
     6	of this script is Docker layer caching: it is COPY'd and RUN in its own
     7	layer BEFORE the app code is copied, so a release that only changes code
     8	does not invalidate the ~60MB model layer (previously the prefetch lived
     9	inside the app layer and every release re-downloaded every model). The
    10	layer re-builds only when THIS FILE changes — i.e. when a model URL/hash
    11	actually changes.
    12	
    13	The (url, sha256, filename) table below MUST stay identical to the
    14	constants in openscrub.py — test_docker_prefetch_matches_engine_pins
    15	fails the suite when they drift.
    16	
    17	Usage:  python prefetch_models.py <key> [<key> ...]     (or no args = all)
    18	"""

    75	def fetch(url, dest, sha256, tries=8, delay=15):
    76	    """Download + sha256-verify with retries (the opencv_zoo LFS media
    77	    host throws transient quota 404s — same policy as _fetch_model)."""
    78	    for attempt in range(1, tries + 1):
    79	        try:
    80	            tmp = dest + ".part"
    81	            with urllib.request.urlopen(url, timeout=120) as r, \
    82	                    open(tmp, "wb") as f:
    83	                while True:
    84	                    chunk = r.read(1 << 20)
    85	                    if not chunk:
    86	                        break
    87	                    f.write(chunk)
    88	            digest = hashlib.sha256(open(tmp, "rb").read()).hexdigest()
    89	            if digest != sha256:
    90	                os.remove(tmp)
    91	                raise RuntimeError(
    92	                    "sha256 mismatch for %s: got %s want %s"
    93	                    % (url, digest, sha256))
    94	            os.replace(tmp, dest)
    95	            print("  ok  %s (%d bytes)" % (dest, os.path.getsize(dest)),
    96	                  flush=True)
    97	            return
    98	        except Exception as e:
    99	            print("  attempt %d/%d failed: %s" % (attempt, tries, e),
   100	                  flush=True)
   101	            if attempt == tries:
   102	                raise
   103	            time.sleep(delay)
   104	
   105	
   106	def main():
   107	    keys = sys.argv[1:] or list(MODELS)
   108	    unknown = [k for k in keys if k not in MODELS]
   109	    if unknown:
   110	        raise SystemExit("unknown model key(s): %s" % ", ".join(unknown))
   111	    os.makedirs(DEST, exist_ok=True)
   112	    for k in keys:
   113	        url, sha, name = MODELS[k]
   114	        print("fetching %s -> %s" % (k, name), flush=True)
   115	        fetch(url, os.path.join(DEST, name), sha)
   116	
   117	
   118	if __name__ == "__main__":
   119	    main()
