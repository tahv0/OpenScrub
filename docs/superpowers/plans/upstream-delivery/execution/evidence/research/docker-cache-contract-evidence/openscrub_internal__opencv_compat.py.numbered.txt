     1	"""Read-only OpenCV fallback checks, independent of models and providers."""
     2	import json
     3	import os
     4	import re
     5	import shlex
     6	import subprocess
     7	
     8	_FLAVORS = ("opencv-python", "opencv-contrib-python", "opencv-python-headless",
     9	            "opencv-contrib-python-headless")
    10	_CAPABILITIES = {"ok", "unavailable", "missing_api", "missing_xml",
    11	                 "classifier_error", "empty_classifier"}
    12	
    13	# Self-contained: a frozen installer may select a Python without OpenScrub.
    14	_PROBE = r'''
    15	# OPENSCRUB_OPENCV_PROBE
    16	import importlib.metadata
    17	import json
    18	import os
    19	result = dict(status="ok", version=None, path=None, distributions={},
    20	              capability="unavailable", detail="")
    21	try:
    22	    import cv2
    23	except ModuleNotFoundError as exc:
    24	    result.update(status="missing_import" if exc.name == "cv2" else "import_error", detail=str(exc))
    25	except Exception as exc:
    26	    result.update(status="import_error", detail=str(exc))
    27	else:
    28	    result.update(version=str(getattr(cv2, "__version__", "")),
    29	                  path=str(getattr(cv2, "__file__", "")))
    30	    if not callable(getattr(cv2, "CascadeClassifier", None)):
    31	        result["capability"] = "missing_api"
    32	    else:
    33	        root = getattr(getattr(cv2, "data", None), "haarcascades", "")
    34	        try:
    35	            if not root:
    36	                raise OSError("cv2.data.haarcascades unavailable")
    37	            xml = os.path.join(root, "haarcascade_frontalface_default.xml")
    38	            with open(xml, "rb") as stream:
    39	                if not stream.read(1):
    40	                    raise OSError("Haar XML is empty")
    41	        except (OSError, TypeError, ValueError) as exc:
    42	            result.update(capability="missing_xml", detail=str(exc))
    43	        else:
    44	            try:
    45	                cascade = cv2.CascadeClassifier(xml)
    46	                result["capability"] = "empty_classifier" if cascade.empty() else "ok"
    47	            except Exception as exc:
    48	                result.update(capability="classifier_error", detail=str(exc))
    49	for name in ("opencv-python", "opencv-contrib-python", "opencv-python-headless", "opencv-contrib-python-headless"):
    50	    try:
    51	        result["distributions"][name] = importlib.metadata.version(name)
    52	    except importlib.metadata.PackageNotFoundError:
    53	        pass
    54	    except Exception as exc:
    55	        result.update(status="metadata_error", detail=str(exc))
    56	print(json.dumps(result))
    57	'''

    59	
    60	def _failure(status, detail):
    61	    return dict(status=status, version=None, path=None, distributions={},
    62	                capability="unavailable", detail=detail)
    63	
    64	
    65	def probe_opencv(python_executable):
    66	    """Collect fresh target-interpreter facts; never import cv2 in this process."""
    67	    try:
    68	        child = subprocess.run([python_executable, "-c", _PROBE],
    69	                               capture_output=True, text=True, timeout=15)
    70	    except subprocess.TimeoutExpired:
    71	        return _failure("timeout", "OpenCV probe exceeded 15 seconds")
    72	    except OSError as exc:
    73	        return _failure("launch_error", str(exc))
    74	    if child.returncode:
    75	        return _failure("crash", "exit %s: %s" % (child.returncode, (child.stderr or "")[-500:]))
    76	    try:
    77	        result = json.loads(child.stdout)
    78	        if not isinstance(result, dict) or set(result) != {
    79	                "status", "version", "path", "distributions", "capability", "detail"}:
    80	            raise ValueError("unexpected probe fields")
    81	        if result["status"] not in {"ok", "missing_import", "import_error", "metadata_error"}:
    82	            raise ValueError("unknown probe status")
    83	        if result["capability"] not in _CAPABILITIES:
    84	            raise ValueError("unknown fallback result")
    85	        if not isinstance(result["distributions"], dict) or any(
    86	                k not in _FLAVORS or not isinstance(v, str)
    87	                for k, v in result["distributions"].items()):
    88	            raise ValueError("invalid distribution metadata")
    89	        if any(v is not None and not isinstance(v, str)
    90	               for v in (result["version"], result["path"])) or not isinstance(result["detail"], str):
    91	            raise ValueError("invalid runtime metadata")
    92	        if result["status"] == "ok" and not result["path"]:
    93	            raise ValueError("missing runtime metadata")
    94	        return result
    95	    except (ValueError, TypeError, KeyError):
    96	        return _failure("malformed_output", "OpenCV probe returned invalid JSON evidence")

    99	def _version(value):
   100	    match = re.fullmatch(r"(\d+)\.(\d+)\.(\d+)(?:\.\d+)?(?:[-+].*)?", value or "")
   101	    return tuple(map(int, match.groups())) if match else None

   104	def assess_opencv(snapshot):
   105	    """Separate fallback capability from wheel ownership and core policy."""
   106	    distributions = snapshot["distributions"]
   107	    usable = snapshot["status"] in {"ok", "metadata_error"} and snapshot["capability"] == "ok"
   108	    runtime = _version(snapshot["version"])
   109	    versions = [_version(v) for v in distributions.values()]
   110	    if len(distributions) > 1:
   111	        state = "conflict"
   112	    elif snapshot["status"] in {"timeout", "crash", "malformed_output", "metadata_error", "launch_error"}:
   113	        state = "uncertain"
   114	    elif snapshot["status"] == "missing_import" and not distributions:
   115	        state = "missing"
   116	    elif any(v and v[0] >= 5 for v in [runtime] + versions):
   117	        state = "unsupported"
   118	    elif any(v is None for v in versions):
   119	        state = "uncertain"
   120	    elif snapshot["status"] in {"import_error", "missing_import"}:
   121	        state = "broken"
   122	    elif runtime is None:
   123	        state = "uncertain"
   124	    elif versions and runtime != versions[0]:
   125	        state = "uncertain"
   126	    elif not usable:
   127	        state = "broken"
   128	    elif not distributions:
   129	        state = "unmanaged"
   130	    elif "opencv-python" not in distributions:
   131	        state = "alternate"
   132	    else:
   133	        state = "healthy"
   134	    evidence = "runtime=%s; path=%s; wheels=%s; fallback=%s; probe=%s" % (
   135	        snapshot["version"] or "unknown", snapshot["path"] or "unknown",
   136	        ", ".join("%s==%s" % pair for pair in distributions.items()) or "none (source/distro/frozen or absent)",
   137	        snapshot["capability"], snapshot["status"])
   138	    if snapshot["detail"]:
   139	        evidence += "; " + snapshot["detail"]
   140	    return dict(state=state, fallback_usable=usable, healthy=state == "healthy", evidence=evidence)
   141	
   142	
   143	def opencv_repair_hint(snapshot, python_executable):
   144	    """Offer normal-wheel repair only with known agreeing runtime/wheel versions."""
   145	    assessment = assess_opencv(snapshot)
   146	    state = assessment["state"]
   147	    command = subprocess.list2cmdline([python_executable]) if os.name == "nt" else shlex.quote(python_executable)
   148	    rerun = "Rerun health checks in a fresh process: %s -m openscrub_setup --check (source: %s install.py --check)." % (command, command)
   149	    if state == "healthy":
   150	        return ""
   151	    runtime = _version(snapshot["version"])
   152	    normal = _version(snapshot["distributions"].get("opencv-python"))
   153	    if (state in {"unsupported", "broken"}
   154	            and set(snapshot["distributions"]) == {"opencv-python"}
   155	            and runtime is not None and runtime == normal):
   156	        return ('To deliberately restore the compatible normal wheel, run: %s -m pip install --force-reinstall "opencv-python<5". ' % command) + rerun
   157	    if state == "missing":
   158	        return ('Install the core requirement with %s -m pip install "opencv-python<5". ' % command) + rerun
   159	    return ("Reconcile cv2 ownership deliberately or use a clean core environment. "
   160	            "Alternate wheels and unmanaged source/distro/frozen builds are not the normal managed wheel; "
   161	            "for source-built CUDA OpenCV, verify the source build includes CascadeClassifier and readable Haar XML. "
   162	            "Do not add another wheel flavor to repair mixed or uncertain ownership. " + rerun)
