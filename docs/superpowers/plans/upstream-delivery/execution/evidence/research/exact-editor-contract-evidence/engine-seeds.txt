  1705	    """Fetch the frame at time t from an open VideoCapture, ROBUSTLY.
  1706	
  1707	    Seeks are NEVER trusted. `cap.set(POS_MSEC/POS_FRAMES)` lies on some
  1708	    codec/build combos: one build FAILS the read outright (h264_nvenc
  1709	    HDR copies in the CUDA image — every deep seek returned nothing, a
  1710	    tracking window silently failed to seed), another SUCCEEDS but lands
  1711	    many seconds early (a real box landed a 20.3s seek at ~4s; the
  1712	    tracker then followed the wrong moment of the video with perfect
  1713	    confidence and the blur 'flew away' off the subject). After every
  1714	    positioning, the decoded packet's own PTS (CAP_PROP_POS_MSEC after
  1715	    read — it comes from the bitstream and cannot lie) is compared to
  1716	    the request; an early landing is repaired by decoding forward, a
  1717	    late one by seeking earlier, ultimately from frame 0 — the
  1718	    renderer's sequential access pattern, correct on any decodable
  1719	    file. Returns the frame or None."""
  1720	    fps = fps or 30.0
  1721	    target = int(round(max(t, 0.0) * fps))
  1722	    st = _grab_state.setdefault(id(cap), {"pos": None, "slow": False})
  1723	
  1724	    def _done(fr):
  1725	        st["pos"] = target + 1          # cap will read target+1 next
  1726	        return fr
  1727

  1729	        # index of the frame just decoded, from the reported packet PTS;
  1730	        # -1 when the build reports nothing usable (then we trust frame
  1731	        # arithmetic instead). POS_MSEC after read() is the NEXT frame's
  1732	        # time on most builds, the just-read frame's on others — the ±1
  1733	        # tolerance at the call sites absorbs the difference.
  1734	        try:
  1735	            ms = float(cap.get(cv2.CAP_PROP_POS_MSEC))
  1736	        except Exception:
  1737	            return -1
  1738	        if ms <= 0:
  1739	            return -1
  1740	        return int(round(ms * fps / 1000.0)) - 1
