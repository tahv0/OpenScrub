  6404	    VFR video; the pipeline's frame->time mapping assumes CFR, so VFR input
  6405	    causes blur-timing drift and audio desync unless normalized first."""

  6409	    try:
  6410	        p = subprocess.run(["ffprobe", "-v", "error", "-select_streams", "v:0",
  6411	                            "-show_entries",
  6412	                            "stream=r_frame_rate,avg_frame_rate",
  6413	                            "-of", "json", path],
  6414	                           capture_output=True, text=True, timeout=30)
  6415	        rc, out = p.returncode, p.stdout
  6416	    except Exception:
  6417	        return False, None
  6418	    if rc != 0 or not out:
  6419	        return False, None
  6420	    try:
  6421	        st = json.loads(out)["streams"][0]
  6422	        def frac(s):
  6423	            a, _, b = s.partition("/")
  6424	            return float(a) / float(b or 1) if float(b or 1) else 0.0
  6425	        r, avg = frac(st.get("r_frame_rate", "0/1")), frac(st.get("avg_frame_rate", "0/1"))

  6496	def _scan_copy_matches(ref, copy, tol=2):
  6497	    """(ok, n_ref, n_copy) — a 1:1 (non-resampling) scan copy must carry
  6498	    exactly the source's frames: the scan reads it by frame INDEX, so a
  6499	    single dropped or duplicated frame silently shifts every detection
  6500	    made after it onto the wrong output frames (misplaced blur — seen on
  6501	    a real GPU box whose NVENC copy desynced). Counting requires a full
  6502	    decode of both files; that cost is nothing next to the scan itself
  6503	    and it is the only check that catches the failure."""
  6504	    n_ref, n_cp = _count_video_frames(ref), _count_video_frames(copy)
  6505	    if n_ref is None or n_cp is None:
  6506	        return True, n_ref, n_cp        # can't measure — don't block
  6507	    return abs(n_ref - n_cp) <= tol, n_ref, n_cp
