   332	def _validate_files_locked(job=None):
   333	    # Admission always rechecks after waiting for the transition guard.
   334	    if vault_locked():
   335	        abort(423, "vault is locked or needs recovery — unlock it to access jobs")
   336	    if job is not None:
   337	        with JOBS_LOCK:
   338	            if not any(current is job for current in JOBS.values()):
   339	                abort(404)
   340	        if FILES.initialize(job).closed:
   341	            abort(409, "job is being deleted")

   344	def _source_job_locked(canonical, missing_ok=False):
   345	    root = os.path.realpath(JOBS_DIR).rstrip(os.sep) + os.sep
   346	    if not canonical.startswith(root):
   347	        return None
   348	    jid = canonical[len(root):].split(os.sep, 1)[0]
   349	    with JOBS_LOCK:
   350	        source_job = JOBS.get(jid)
   351	    if source_job is None:
   352	        if missing_ok:
   353	            return None
   354	        abort(404, "source job is no longer available")
   355	    _validate_files_locked(source_job)
   356	    return source_job

   359	def _reserve_files_locked(job=None, work=False, read_source=True, source=None):
   360	    source = (os.path.realpath(source if source is not None else job["video"])
   361	              if read_source and job and (source is not None or job.get("video")) else None)
   362	    source_job = _source_job_locked(source) if source else None
   363	    lease = FILES.reserve(job, work)
   364	    lease.source = source
   365	    if source_job is not None and source_job is not job:
   366	        lease.related.append(FILES.reserve(source_job))
   367	    return lease

   444	def _update_job_source(job, path, owner):
   445	    canonical = os.path.realpath(path)
   446	    with VAULT_LOCK:
   447	        _validate_files_locked(job)
   448	        source_job = _source_job_locked(canonical)
   449	        if source_job is not None and source_job is not job:
   450	            # Reserve before following a changed normalized input. This stays
   451	            # with the worker through its final report/provenance access.
   452	            if owner is not None and all(
   453	                    token.state is not FILES.initialize(source_job)
   454	                    for token in [owner] + owner.related):
   455	                owner.related.append(FILES.reserve(source_job))
   456	        job["video"] = canonical

   495	def _new_job(video_path, options, name):
   496	    jid = uuid.uuid4().hex[:12]
   497	    jdir = os.path.join(JOBS_DIR, jid)
   498	    os.makedirs(jdir, exist_ok=True)
   499	    job = {
   500	        "id": jid, "dir": jdir, "video": video_path, "name": name,
   501	        "options": options, "phase": "queued", "progress": 0.0,
   502	        "eta": None, "size": 0,
   503	        "log": [], "error": None, "output": None, "created": time.time(),
   504	        "cancel": threading.Event(), "stats": {}, "qgen": 0,
   505	    }
   506	    FILES.initialize(job)
   507	    # keep the job's settings beside its upload: a scan that never finished
   508	    # has no report to recover them from, and a Retry after a restart then
   509	    # ran with DEFAULT settings (all categories) instead of the user's own
   510	    try:
   511	        with open(os.path.join(jdir, OPTIONS_FILE), "w",
   512	                  encoding="utf-8") as f:
   513	            json.dump(options, f)
   514	    except (OSError, TypeError, ValueError):
   515	        pass
   516	    with VAULT_LOCK:
   517	        with JOBS_LOCK:
   518	            JOBS[jid] = job
   519	        if has_request_context() and g.get("file_owner") is not None:
   520	            g.setdefault("file_children", []).append(FILES.reserve(job, work=True))
   521	    return job
