  1703	    """Only immutable, finite creation records can establish replay equality."""
  1704	    def numbers(items, size):
  1705	        try:
  1706	            return (isinstance(items, list) and len(items) == size
  1707	                    and all(type(v) in (int, float) and math.isfinite(v)
  1708	                            for v in items))
  1709	        except OverflowError:
  1710	            return False
  1711	    return (isinstance(value, dict)
  1712	            and set(value) == {"t_start", "t_end", "cbox", "aoff"}
  1713	            and numbers([value["t_start"], value["t_end"]], 2)
  1714	            and numbers(value["cbox"], 4)
  1715	            and all(type(v) is int for v in value["cbox"])
  1716	            and numbers(value["aoff"], 2))
