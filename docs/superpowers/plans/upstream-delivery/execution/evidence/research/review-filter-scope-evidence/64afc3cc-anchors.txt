 5347  function reviewPreviewRank(task){
 5348   if(!task.img.isConnected||task.img.closest?.("[hidden]"))return Infinity;
 5349   if(task.priority)return -1; // An explicit retry of a visible image gets priority.
 5350   // Old non-layout harnesses / browsers without geometry retain FIFO behavior.
 5351   if(!task.img.getBoundingClientRect)return 0;
 5352   const r=task.img.getBoundingClientRect(),h=window.innerHeight||document.documentElement.clientHeight;
 5353   // A card far BEHIND the viewport is obsolete until the user scrolls back.
 5354   // Cards AHEAD stay eligible for speculative JPEG cache reads even when they
 5355   // are far beyond IntersectionObserver's margin. Scroll changes re-rank work
 5356   // at dispatch: visible cards overtake the background queue.
 5357   if(r.bottom < -REVIEW_STALE_BEHIND_PX)return Infinity;
 5358   return r.bottom<0?-r.bottom:r.top>h?r.top-h:0;
 5359  }
 5360  function pumpReviewPreviews(state){
 5361   if(REVIEW_MEDIA!==state)return;
 5362   if(state.retryTimer){clearTimeout(state.retryTimer);state.retryTimer=null;}
 5363   let wake=Infinity;
 5364   // Cache hits overtake cold decoding. Re-rank at dispatch, not when the card
 5365   // first entered view: scrolling must not leave minutes of obsolete work.
 5366   for(const [queue,counter,limit,cached] of [
 5367    [state.cacheQueue,"cacheActive",3,true],[state.queue,"active",1,false]]){
 5368    const ranked=queue.splice(0).map(task=>({task,rank:reviewPreviewRank(task)}));
 5369    ranked.sort((a,b)=>a.rank-b.rank);
 5370    for(const {task,rank} of ranked){
 5371     if(rank===Infinity){task.img._previewTask=null;task.resolve(false);}
 5372     else queue.push(task);
 5373    }
 5374    while(state[counter]<limit&&queue.length){
 5375     const n=queue.findIndex(task=>
 5376      (!task.readyAt||task.readyAt<=Date.now()) &&
 5377      (cached||reviewPreviewRank(task)<=REVIEW_COLD_LOOKAHEAD_PX));
 5378     if(n<0)break;
 5379     loadReviewPreview(state,queue.splice(n,1)[0],cached);
 5380    }

 5382   }
