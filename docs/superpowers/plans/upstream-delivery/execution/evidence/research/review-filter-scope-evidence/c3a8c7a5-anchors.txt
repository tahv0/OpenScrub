 6102  function reviewGroupStatus(i){
 6103   const members=TRKMEM[i]||[i];
 6104   const handled=members.some(m=>!!OUTPUT_CHOICES[m]||!!reviewFindingLabel(m));
 6105   const pending=members.some(m=>{
 6106    const edit=PLATE_EDITS.get(m);
 6107    return !!((edit&&(edit.dirty||edit.pending||edit.drag))||CONFIRM_PENDING.has(m)||CONFIRM_FAILED.has(m));
 6108   });
 6109   return {handled,pending};
 6110  }
 6111  function reviewMatchesFilter(i,cat){
 6112   if(LEARNING_SELECTION&&(cat!=="plate"||!(TRKMEM[i]||[i]).some(m=>LEARNING_SELECTION.has(m))))return false;
 6113   const status=reviewGroupStatus(i);
 6114   if(REVIEW_STATUS_FILTER==="unreviewed")return !status.handled||status.pending;
 6115   if(REVIEW_STATUS_FILTER==="reviewed")return status.handled&&!status.pending;
 6116   if(REVIEW_STATUS_FILTER==="pending")return status.pending;
 6117   return true;
 6118  }

 6134  function applyReviewGroupFilter(){

 6136   if(!panel||panel._sampleJob!==CUR||panel._sampleView!==JOBVIEW)return false;
 6137   const selected=LEARNING_SELECTION;
 6138   const totals={all:0,unreviewed:0,reviewed:0,pending:0};
 6139   for(const indices of Object.values(REVIEW_CATEGORY_GROUPS)){
 6140    for(const i of indices){
 6141     const group=reviewGroupStatus(i);totals.all++;
 6142     if(group.pending)totals.pending++;
 6143     if(!group.handled||group.pending)totals.unreviewed++;
 6144     else totals.reviewed++;
 6145    }

 6159    if(!card.hidden)shown++;

 6161   panel.querySelectorAll("[data-review-category]").forEach(grid=>{
 6162    const cat=grid.dataset.reviewCategory;

 6164    grid.hidden=matching===0;

 6171    button.hidden=!!selected||pending===0;
 6172    button.textContent="Load more "+(CATDN[cat]||cat)+" ("+pending+" remaining)";
 6173   });

 6175   const all=panel.querySelector("#reviewShowAll");if(all)all.hidden=!selected&&REVIEW_STATUS_FILTER==="all";
