   754	class PhiMemory:
   755	    """Remembers every string ever confirmed as PII in this video. At each
   756	    scan, all OCR'd words are checked against memory, so a name identified
   757	    once gets blurred on every later appearance — anywhere on screen — even
   758	    when NER/heuristics fail on that occurrence. Alpha strings match fuzzily
   759	    (handles OCR misreads); numeric strings require same length with at most
   760	    one differing digit (so benign numbers don't collide with MRNs)."""
   761	
   762	    IMMEDIATE = {"dob", "phone", "ssn", "email", "mrn", "address",
   763	                 "qrcode", "screen", "anytext", "phrase",
   764	                 "apikey", "ipaddr", "card", "plate", "person"}
   765	
   766	    def __init__(self, threshold=82, name_sightings=2):
   767	        from rapidfuzz import fuzz
   768	        self.fuzz = fuzz
   769	        self.threshold = threshold
   770	        self.name_sightings = name_sightings
   771	        self.items = {}    # normalized text -> category
   772	        self.counts = {}   # normalized text -> primary-detector sightings
   773	        # Recall used to scan every remembered string for every OCR word.
   774	        # Long recordings therefore got progressively slower as memory grew.
   775	        # Length buckets (contributed by Tuomas Aho) keep each search local
   776	        # to plausible candidates, and every bucket is matched in ONE
   777	        # compiled rapidfuzz batch call instead of a Python loop — the
   778	        # per-string fuzzy comparisons were the real cost. Matching rules
   779	        # are unchanged: same scores, same thresholds, and the FIRST
   780	        # remembered match still wins (lowest insertion serial).
   781	        self._alpha = {}    # length -> ([text], [insertion serial])
   782	        self._digits = {}   # length -> [text], insertion-ordered
   783	        self._memory_serial = 0
   784	
   785	    @staticmethod
   786	    def norm(s):
   787	        return s.strip(".,:;()[]").lower()
   788	
   789	    def add(self, text, category, primary=True):
   790	        if category in ("face", "manual"):
   791	            return
   792	        n = self.norm(text)
   793	        if len(n) >= 3 and n not in STOPWORDS:
   794	            if n not in self.items:
   795	                self.items[n] = category
   796	                if n.isdigit():
   797	                    self._digits.setdefault(len(n), []).append(n)
   798	                else:
   799	                    keys, serials = self._alpha.setdefault(len(n), ([], []))
   800	                    keys.append(n)
   801	                    serials.append(self._memory_serial)
   802	                self._memory_serial += 1
   803	            if primary:
   804	                self.counts[n] = self.counts.get(n, 0) + 1
   805	
   806	    def _gated(self, key, cat):
   807	        """Names must be seen by a primary detector on name_sightings
   808	        separate scans before memory starts recalling them — one bad
   809	        NER hit shouldn't multiply across the whole video. Regex
   810	        categories are high-precision and recall immediately."""
   811	        if cat in self.IMMEDIATE:
   812	            return cat
   813	        return cat if self.counts.get(key, 0) >= self.name_sightings else None
   814	
   815	    def recall(self, word):
   816	        n = self.norm(word)
   817	        if len(n) < 3 or n in STOPWORDS:
   818	            return None
   819	        if n in self.items:
   820	            return self._gated(n, self.items[n])
   821	        from rapidfuzz import process
   822	        from rapidfuzz.distance import Hamming
   823	        if n.isdigit():
   824	            # same length, at most one differing digit; first remembered wins
   825	            keys = self._digits.get(len(n))
   826	            if keys:
   827	                hits = process.extract(n, keys, scorer=Hamming.distance,
   828	                                       score_cutoff=1, limit=None,
   829	                                       processor=None)
   830	                if hits:
   831	                    k = keys[min(h[2] for h in hits)]
   832	                    return self._gated(k, self.items[k])
   833	            return None
   834	        if len(n) >= 4:
   835	            best = None                     # (insertion serial, text)
   836	            for size in range(max(3, len(n) - 2), len(n) + 3):
   837	                bucket = self._alpha.get(size)
   838	                if not bucket:
   839	                    continue
   840	                keys, serials = bucket
   841	                for _, _, i in process.extract(
   842	                        n, keys, scorer=self.fuzz.ratio,
   843	                        score_cutoff=self.threshold, limit=None,
   844	                        processor=None):
   845	                    if best is None or serials[i] < best[0]:
   846	                        best = (serials[i], keys[i])
   847	            if best is not None:
   848	                return self._gated(best[1], self.items[best[1]])
   849	        return None

  1327	def boxes_overlap(a, b, slack=12):
  1328	    return not (a[2] + slack < b[0] or b[2] + slack < a[0]
  1329	                or a[3] + slack < b[1] or b[3] + slack < a[1])

  2915	def merge_detections(dets, hold, scans=None, bridge_gap=4.0, fuzz=None,
  2916	                     gap_check=None):
  2917	    """Chain detections of the same category whose content boxes overlap.
  2918	
  2919	    Short gaps (within `hold`) chain unconditionally, as before. Longer gaps
  2920	    up to `bridge_gap` seconds are BRIDGED — kept blurred straight through —
  2921	    unless an intermediate scan positively saw different, readable text in
  2922	    that region (i.e. the content genuinely changed). An empty or unreadable
  2923	    region during the gap is treated as an OCR miss and stays covered:
  2924	    fail closed, never flash PII."""
  2925	    dets = sorted(dets, key=lambda d: d.t_start)
  2926	    merged = []
  2927	    mergeable = []  # Non-dense detections only
  2928	
  2929	    def contradicted(m, t_from, t_to):
  2930	        """True only if the gap contains STABLE different text — the same
  2931	        different string read on two or more scans. A single divergent read
  2932	        is far more likely to be the mouse cursor sitting over the word (or
  2933	        another transient occlusion) garbling OCR than genuinely new content:
  2934	        real replacement text reads consistently, cursor garble varies every
  2935	        scan. Fail closed — an unstable read keeps the region blurred."""
  2936	        if not scans:
  2937	            return False
  2938	        mx1, my1, mx2, my2 = m.cbox
  2939	        seen = {}
  2940	        for st, _cum, words in scans:
  2941	            if not (t_from + 0.01 < st < t_to - 0.01):
  2942	                continue
  2943	            for txt, (x1, y1, x2, y2), conf in words:
  2944	                if conf < 0.6:
  2945	                    continue
  2946	                cxm = (x1 + x2) / 2
  2947	                cym = (y1 + y2) / 2
  2948	                if mx1 - 6 <= cxm <= mx2 + 6 and my1 - 4 <= cym <= my2 + 4:
  2949	                    n = PhiMemory.norm(txt)
  2950	                    if fuzz:
  2951	                        mt = PhiMemory.norm(m.text)
  2952	                        # partial_ratio catches cursor-occluded reads of the
  2953	                        # SAME word ("errin" ~ "herrin"): those are evidence
  2954	                        # the word is still there, never evidence it changed
  2955	                        same = (fuzz.ratio(n, mt) >= 70
  2956	                                or fuzz.partial_ratio(n, mt) >= 85)
  2957	                    else:
  2958	                        same = False
  2959	                    if not same and len(n) >= 3:
  2960	                        seen[n] = seen.get(n, 0) + 1
  2961	                        if seen[n] >= 2:
  2962	                            return True
  2963	        return False
  2964	
  2965	    for d in dets:
  2966	        d.last_seen = d.t_start
  2967	        if getattr(d, 'dense', False):
  2968	            merged.append(d)
  2969	            continue
  2970	        if not getattr(d, "dense", False):
  2971	            # dense samples keep their sub-frame hold: stamping them with the
  2972	            # multi-second OCR hold leaves every PAST position blurred for
  2973	            # `hold` seconds — a trail of stale boxes marching away from a
  2974	            # moving face. Their continuity across detector flicker and the
  2975	            # onset gap is handled per-track by smooth_dense_tracks().
  2976	            d.t_end = d.t_start + hold
  2977	        for m in reversed(mergeable):
  2978	            if m.category != d.category or not boxes_overlap(m.cbox, d.cbox):
  2979	                continue
  2980	            if getattr(d, "dense", False) or getattr(m, "dense", False):
  2981	                # dense face boxes are per-frame position samples of a possibly
  2982	                # moving face — never merge them, or the bounding box balloons
  2983	                # to cover the whole path. Each stands alone with its short
  2984	                # hold, so the blur rides the face frame by frame.
  2985	                continue
  2986	            # different readable text in the same spot is a DIFFERENT object
  2987	            # (e.g. names sliding through one row of an inner-scrolling list)
  2988	            # — never fuse them, or the region's text and its frames diverge
  2989	            if fuzz and m.text and d.text:
  2990	                _mn = PhiMemory.norm(m.text)
  2991	                _dn = PhiMemory.norm(d.text)
  2992	                if (fuzz.ratio(_mn, _dn) < 70
  2993	                        and fuzz.partial_ratio(_mn, _dn) < 85):
  2994	                    continue
  2995	            gap = d.t_start - m.last_seen
  2996	            ok = gap <= hold
  2997	            if not ok and not contradicted(m, m.last_seen, d.t_start):
  2998	                if gap <= bridge_gap:
  2999	                    ok = True
  3000	                elif gap_check is not None:
  3001	                    # beyond the configured bridge: ask the pixels. The file
  3002	                    # is checked at points inside the gap — bridge any length
  3003	                    # of gap the content verifiably persisted through, refuse
  3004	                    # if it visibly changed. The knob stops mattering.
  3005	                    ok = gap_check(m, m.last_seen, d.t_start)
  3006	            if ok:
  3007	                m.last_seen = max(m.last_seen, d.t_start)
  3008	                m.t_end = max(m.t_end, d.t_end)
  3009	                m.cbox = (min(m.cbox[0], d.cbox[0]), min(m.cbox[1], d.cbox[1]),
  3010	                          max(m.cbox[2], d.cbox[2]), max(m.cbox[3], d.cbox[3]))
  3011	                break
  3012	        else:
  3013	            merged.append(d)
  3014	            mergeable.append(d)
  3015	
  3016	    # Tail extension: a span ends `hold` after its LAST positive detection,
  3017	    # but the text often outlives that detection — under heavy compression
  3018	    # the final scans read only fragments ('4111.' '1111-1111' '1111'),
  3019	    # which fail the full pattern yet are strong evidence the string is
  3020	    # still on screen (the benchmark's compressed card leaked exactly this
  3021	    # tail). Walk the scans PAST each span's last sighting: while a scan's
  3022	    # words in the region fuzzy-match the span text (the same same-word
  3023	    # test contradicted() uses), the content verifiably persisted — extend
  3024	    # coverage through that scan. A scan with no matching read stops the
  3025	    # walk, so a real scene change never accretes stale blur.
  3026	    if scans and fuzz:
  3027	        for m in merged:
  3028	            if getattr(m, "dense", False) or not m.text:
  3029	                continue
  3030	            mx1, my1, mx2, my2 = m.cbox
  3031	            mt = PhiMemory.norm(m.text)
  3032	            for st, _cum, words in scans:
  3033	                if st <= m.last_seen + 0.01:
  3034	                    continue
  3035	                if st - m.last_seen > bridge_gap:
  3036	                    break
  3037	                support = False
  3038	                for txt, (x1, y1, x2, y2), conf in words:
  3039	                    if conf < 0.5:
  3040	                        continue
  3041	                    cxm = (x1 + x2) / 2
  3042	                    cym = (y1 + y2) / 2
  3043	                    if not (mx1 - 6 <= cxm <= mx2 + 6
  3044	                            and my1 - 4 <= cym <= my2 + 4):
  3045	                        continue
  3046	                    n = PhiMemory.norm(txt)
  3047	                    if len(n) >= 3 and (fuzz.ratio(n, mt) >= 70
  3048	                                        or fuzz.partial_ratio(n, mt) >= 85):
  3049	                        support = True
  3050	                        break
  3051	                if not support:
  3052	                    break
  3053	                m.last_seen = st
  3054	                m.t_end = max(m.t_end, st + hold)
  3055	    return merged

  7097	    """After the scan, re-search every OCR'd word against remembered PII at
  7098	    a more lenient threshold. Catches near-misses (OCR misreads) of strings
  7099	    already confirmed elsewhere in the video; gating rules still apply so a
  7100	    one-off false positive can't spread."""
  7101	    from rapidfuzz import fuzz
  7102	    extra = []
  7103	    for t, cum, words in scans:
  7104	        for w, cbox, conf in words:
  7105	            n = PhiMemory.norm(w)
  7106	            if len(n) < 4 or n in STOPWORDS:
  7107	                continue
  7108	            if namer and namer._allowed(w):
  7109	                continue
  7110	            for k, cat in memory.items.items():
  7111	                if cat not in cats:
  7112	                    continue
  7113	                if memory._gated(k, cat) is None:
  7114	                    continue
  7115	                if k.isdigit() or n.isdigit():
  7116	                    hit = (k.isdigit() and n.isdigit() and len(k) == len(n)
