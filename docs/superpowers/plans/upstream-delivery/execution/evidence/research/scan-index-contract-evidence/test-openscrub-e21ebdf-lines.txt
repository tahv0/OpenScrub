   135	def test_batched_memory_recall_matches_reference_scan():
   136	    """The indexed + batched recall must return EXACTLY what the original
   137	    linear scan returned (same scores, thresholds, first-remembered tie
   138	    break, name gating) — a privacy tool can't trade recall for speed.
   139	    Differential check against a verbatim copy of the old algorithm."""
   140	    import random
   141	    import string
   142	    from rapidfuzz import fuzz
   143	
   144	    def reference(m, word):
   145	        n = m.norm(word)
   146	        if len(n) < 3 or n in openscrub.STOPWORDS:
   147	            return None
   148	        if n in m.items:
   149	            return m._gated(n, m.items[n])
   150	        if n.isdigit():
   151	            for k, cat in m.items.items():
   152	                if (k.isdigit() and len(k) == len(n)
   153	                        and sum(a != b for a, b in zip(k, n)) <= 1):
   154	                    return m._gated(k, cat)
   155	            return None
   156	        if len(n) >= 4:
   157	            for k, cat in m.items.items():
   158	                if (not k.isdigit() and abs(len(k) - len(n)) <= 2
   159	                        and fuzz.ratio(k, n) >= m.threshold):
   160	                    return m._gated(k, cat)
   161	        return None
   162	
   163	    rng = random.Random(7)
   164	
   165	    def mutate(s):
   166	        s = list(s)
   167	        for _ in range(rng.randint(0, 3)):
   168	            i = rng.randrange(len(s))
   169	            op = rng.random()
   170	            if op < 0.4:
   171	                s[i] = rng.choice("abcdeh0123")
   172	            elif op < 0.7:
   173	                s.insert(i, rng.choice("abcdeh"))
   174	            elif len(s) > 3:
   175	                s.pop(i)
   176	        return "".join(s)
   177	
   178	    for thr in (50, 82, 90):
   179	        m = openscrub.PhiMemory(threshold=thr)
   180	        pool = []
   181	        for _ in range(300):        # tiny alphabet => many ties
   182	            if pool and rng.random() < 0.3:
   183	                w = mutate(rng.choice(pool))
   184	            elif rng.random() < 0.3:
   185	                w = "".join(rng.choice(string.digits)
   186	                            for _ in range(rng.randint(3, 11)))
   187	            else:
   188	                w = "".join(rng.choice("abcdeh")
   189	                            for _ in range(rng.randint(3, 12)))
   190	            m.add(w, rng.choice(["name", "phone", "ssn", "plate"]),
   191	                  primary=rng.random() < 0.7)
   192	            pool.append(w)
   193	        for _ in range(400):
   194	            q = mutate(rng.choice(pool))
   195	            assert m.recall(q) == reference(m, q), (thr, q)
